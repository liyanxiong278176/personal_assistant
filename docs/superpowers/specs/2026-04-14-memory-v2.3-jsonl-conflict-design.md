# Memory System v2.3 Design Spec

> 补齐JSONL持久化 + 冲突策略完善，让代码完全匹配STAR描述

**Goal:** 补齐记忆管理系统缺失的两项功能：
1. JSONL持久化：语义记忆写入时同步备份到JSONL文件
2. 冲突策略完善：增加OVERWRITE和CLEAR两种策略

**Architecture:**
- JSONL备份：同步写入，不阻塞主流程，失败只warning
- OVERWRITE：LLM判定触发，完全替换旧记忆
- CLEAR：LLM判定触发，批量清除某类记忆

### 1.1 实现位置

### 1.2 核心类设计

```python
@dataclass
class SemanticBackupConfig:
    """JSONL backup configuration."""
    enabled: bool = True
    backup_path: str = "./data/semantic_backup.jsonl"
    max_file_size_mb: float = 100.0  # 超过此大小触发rotate


class SemanticJSONLBackup:
    """JSONL backup for semantic memories.

    Writes each semantic memory to a JSONL file as backup.
    Synchronous write, non-blocking on failure (only warning).
    """

    def __init__(self, config: SemanticBackupConfig):
        self._config = config
        self._ensure_directory()

    def append(self, memory: MemoryItem) -> bool:
        """Append a memory to JSONL file.

        Args:
            memory: MemoryItem to backup

        Returns:
            True if written successfully, False otherwise
        """
        if not self._config.enabled:
            return True

        try:
            line = json.dumps(memory.to_dict(), ensure_ascii=False)
            with open(self._config.backup_path, "a", encoding="utf-8") as f:
                f.write(line + "\n")
            return True
        except Exception as e:
            logger.warning(f"[SemanticBackup] Write failed: {e}")
            return False

    def read_all(self) -> List[MemoryItem]:
        """Read all memories from backup file.

        Used for recovery when ChromaDB data lost.
        """
        if not os.path.exists(self._config.backup_path):
            return []

        memories = []
        skipped_count = 0
        with open(self._config.backup_path, "r", encoding="utf-8") as f:
            for line_num, line in enumerate(f, 1):
                try:
                    data = json.loads(line.strip())
                    memories.append(MemoryItem.from_dict(data))
                except json.JSONDecodeError as e:
                    skipped_count += 1
                    logger.warning(
                        f"[SemanticBackup] Skip corrupted line {line_num}: {str(e)[:50]}"
                    )

        if skipped_count > 0:
            logger.warning(
                f"[SemanticBackup] Recovery complete | "
                f"loaded={len(memories)} | skipped={skipped_count}"
            )

        return memories

    def get_stats(self) -> dict:
        """Get backup file statistics."""
        if not os.path.exists(self._config.backup_path):
            return {"exists": False, "count": 0}

        size_mb = os.path.getsize(self._config.backup_path) / (1024 * 1024)
        count = sum(1 for _ in open(self._config.backup_path, "r", encoding="utf-8"))
        return {
            "exists": True,
            "path": self._config.backup_path,
            "size_mb": round(size_mb, 2),
            "count": count,
            "max_size_mb": self._config.max_file_size_mb,
        }
```

## 2. 冲突策略完善

### 2.1 操作类型扩展

**修改 `backend/app/core/memory/conflict_resolver.py`：**

```python
class MemoryOperation(Enum):
    """Memory operation types."""
    ADD = "add"         # 全新信息，无相似记忆
    UPDATE = "update"   # 同类信息merge（保留metadata）
    OVERWRITE = "overwrite"  # 完全替换（删除旧记忆，新建）
    DELETE = "delete"         # 单条记忆失效
    CLEAR = "clear"           # 批量清除某类记忆
    NOOP = "noop"             # 重复/忽略
```

### 2.2 LLM Prompt更新

```python
CONFLICT_PROMPT = """
判断以下两句话的关系：

旧信息: {existing_content}
新信息: {new_content}
旧信息类型: {existing_type}

请回答以下选项之一：
- UPDATE: 新信息是对旧信息的更新或修正（合并保留细节）
- OVERWRITE: 新信息完全替代旧信息（用户明确否定旧偏好）
- NOOP: 新信息与旧信息一致或重复
- DELETE: 新信息表示旧信息已失效
- CLEAR: 用户请求清除相关记忆（回答格式: CLEAR:类型，如CLEAR:preference）

只回答选项名称，不要解释。
"""
```

### 2.2.1 LLM响应解析

```python
def _parse_llm_response(self, response: str, existing_item: MemoryItem) -> tuple[MemoryOperation, Optional[MemoryType]]:
    """Parse LLM response to get operation and optional clear type.

    Returns:
        (operation, clear_type) tuple
    """
    response = response.strip().upper()

    # CLEAR response may include type: "CLEAR:PREFERENCE" or just "CLEAR"
    if response.startswith("CLEAR"):
        parts = response.split(":")
        if len(parts) > 1:
            try:
                clear_type = MemoryType(parts[1].lower())
            except ValueError:
                clear_type = existing_item.memory_type  # fallback to existing type
        else:
            clear_type = existing_item.memory_type  # clear the matched type
        return (MemoryOperation.CLEAR, clear_type)

    # Other operations
    operation_map = {
        "UPDATE": MemoryOperation.UPDATE,
        "OVERWRITE": MemoryOperation.OVERWRITE,
        "DELETE": MemoryOperation.DELETE,
        "NOOP": MemoryOperation.NOOP,
        "ADD": MemoryOperation.ADD,
    }

    first_word = response.split()[0] if response.split() else response
    return (operation_map.get(first_word, MemoryOperation.NOOP), None)
```

### 2.3 Resolution处理逻辑

```python
def _apply_resolution(self, result: ConflictResolution) -> Optional[MemoryItem]:
    """Apply resolution and return the item to store."""
    if result.operation == MemoryOperation.ADD:
        return result.new_item

    elif result.operation == MemoryOperation.UPDATE:
        # Merge: keep existing metadata, update content
        if result.existing_item and result.new_item:
            merged = result.existing_item
            merged.content = result.new_item.content
            merged.metadata.update(result.new_item.metadata or {})
            return merged
        return result.new_item

    elif result.operation == MemoryOperation.OVERWRITE:
        # 完全替换：删除旧记忆，返回新记忆（不merge）
        # 上层需要先删除existing_item，再add new_item
        return result.new_item

    elif result.operation == MemoryOperation.DELETE:
        # 单条删除：返回None表示不存储
        return None

    elif result.operation == MemoryOperation.CLEAR:
        # 批量清除：返回特殊标记，上层按类型清除
        # 体现在ConflictResolution.clear_type字段
        return None

    else:  # NOOP
        return result.existing_item
```

### 2.4 ConflictResolution数据结构扩展

```python
@dataclass
class ConflictResolution:
    """Conflict resolution result."""
    operation: MemoryOperation = MemoryOperation.ADD
    existing_item: Optional[MemoryItem] = None
    new_item: Optional[MemoryItem] = None
    reason: str = ""
    similarity: float = 0.0
    fallback_used: bool = False
    has_conflict: bool = False
    resolved_item: Optional[MemoryItem] = None
    # v2.3新增：CLEAR操作的类型范围
    clear_type: Optional[MemoryType] = None
```

| 场景 | 用户输入 | 旧记忆 | LLM判定 |

### 2.6 上层调用处理（hierarchy.py集成）

`MemoryHierarchy.add_semantic_with_conflict_check()` 方法需要处理OVERWRITE和CLEAR：

```python
# backend/app/core/memory/hierarchy.py - add_semantic_with_conflict_check() 修改

async def add_semantic_with_conflict_check(
    self,
    item: MemoryItem,
    resolve: bool = True,
) -> "ConflictResolution":
    """Add semantic memory with conflict detection and resolution.

    v2.3新增: 处理OVERWRITE和CLEAR操作
    """
    if item.level != MemoryLevel.SEMANTIC:
        item.level = MemoryLevel.SEMANTIC

    async with self._semantic_lock:
        if self._conflict_resolver:
            conflict_result = await self._conflict_resolver.check_conflict(
                item, self._semantic
            )

            if conflict_result.has_conflict and resolve:
                # v2.3: 根据operation类型执行不同操作
                if conflict_result.operation == MemoryOperation.OVERWRITE:
                    # OVERWRITE: 先删除旧记忆，再添加新记忆
                    if conflict_result.existing_item:
                        self._semantic = [
                            m for m in self._semantic
                            if m.item_id != conflict_result.existing_item.item_id
                        ]
                        logger.info(
                            f"[MemoryHierarchy] OVERWRITE: removed {conflict_result.existing_item.item_id}"
                        )
                    # 添加新记忆（不merge旧metadata）
                    new_item = MemoryItem(
                        content=item.content,
                        level=MemoryLevel.SEMANTIC,
                        memory_type=item.memory_type,
                        metadata=item.metadata or {},  # 不merge，使用新metadata
                        importance=item.importance,
                    )
                    self._semantic.append(new_item)
                    conflict_result.resolved_item = new_item

                elif conflict_result.operation == MemoryOperation.CLEAR:
                    # CLEAR: 按类型批量清除
                    clear_type = conflict_result.clear_type
                    if clear_type:
                        removed_count = len([
                            m for m in self._semantic
                            if m.memory_type == clear_type
                        ])
                        self._semantic = [
                            m for m in self._semantic
                            if m.memory_type != clear_type
                        ]
                        logger.info(
                            f"[MemoryHierarchy] CLEAR: removed {removed_count} memories of type {clear_type}"
                        )
                    conflict_result.resolved_item = None

                else:
                    # 其他操作（UPDATE/DELETE/NOOP）由resolver处理
                    resolved_item = self._conflict_resolver._apply_resolution(conflict_result)
                    if resolved_item:
                        self._semantic.append(resolved_item)

                return conflict_result

        # 无冲突：直接添加
        self._semantic.append(item)
        from app.core.memory.conflict_resolver import ConflictResolution
        return ConflictResolution(has_conflict=False, resolved_item=item)
```

import pytest
import os
import tempfile
from app.core.memory.semantic_backup import SemanticJSONLBackup, SemanticBackupConfig
from app.core.memory.hierarchy import MemoryItem, MemoryLevel, MemoryType

def test_append_and_read():
    """Test writing and reading from JSONL backup."""
    with tempfile.NamedTemporaryFile(suffix=".jsonl", delete=False) as f:
        config = SemanticBackupConfig(backup_path=f.name)
        backup = SemanticJSONLBackup(config)

        # Write two memories
        mem1 = MemoryItem(
            content="喜欢自然风光",
            level=MemoryLevel.SEMANTIC,
            memory_type=MemoryType.PREFERENCE,
            importance=0.8,
        )
        mem2 = MemoryItem(
            content="预算5000元",
            level=MemoryLevel.SEMANTIC,
            memory_type=MemoryType.CONSTRAINT,
            importance=0.9,
        )

        assert backup.append(mem1) is True
        assert backup.append(mem2) is True

        # Read back
        memories = backup.read_all()
        assert len(memories) == 2
        assert memories[0].content == "喜欢自然风光"
        assert memories[1].memory_type == MemoryType.CONSTRAINT

        os.unlink(f.name)

def test_disabled_backup():
    """Test that disabled backup returns True without writing."""
    config = SemanticBackupConfig(enabled=False)
    backup = SemanticJSONLBackup(config)

    mem = MemoryItem(content="test", level=MemoryLevel.SEMANTIC)
    assert backup.append(mem) is True

def test_get_stats():
    """Test backup file statistics."""
    with tempfile.NamedTemporaryFile(suffix=".jsonl", delete=False) as f:
        config = SemanticBackupConfig(backup_path=f.name)
        backup = SemanticJSONLBackup(config)

        # Empty file stats
        stats = backup.get_stats()
        assert stats["exists"] is True

## 6. 验收标准

完成所有任务后，应满足：

1. **JSONL备份**: 每条语义记忆写入后同步备份，失败不阻塞
2. **备份恢复**: `read_all()`可读取全部备份用于恢复
3. **OVERWRITE**: LLM返回OVERWRITE时，旧记忆完全删除，新记忆独立创建
4. **CLEAR**: LLM返回CLEAR时，返回清除类型，上层批量删除
5. **策略完整**: 6种策略（ADD/UPDATE/OVERWRITE/DELETE/CLEAR/NOOP）覆盖所有场景
6. **测试覆盖**: 新增功能有对应单元测试
